# GPIO for the indicators, PIR edge and (on a Pi 5, with the lgpio
# backend) the RC522 CS line. The PIR's real driver is a gpiozero
# Button with bounce_time; the indicators are gpiozero LEDs.
#   export GPIOZERO_PIN_FACTORY=lgpio
gpiozero>=2.0
lgpio>=0.2.2.0

# RC522 RFID over /dev/spidev0.0 (modules/rfid.py).
spidev>=3.6

# For the DHT11's microsecond edge timestamps (modules/dht11.py). The
# OS cannot bit-bang in userspace; pigpio samples edges in hardware.
pigpio>=1.78

# Optional, only for the TODO bodies in modules/rtc.py and wifi.py —
# urllib3 is already in the stdlib's urllib, so wifi.py needs nothing.
